## 1. Two further linear consequences of saturation

Notation of the Stage A report. Suppose `c_0 = r_in = r`. Fix a facet
normal `n`, the edge direction `e = Rot90(n)`, and `x_+`, `x_-` the unit
eigenvectors of `n.sigma` (eigenvalues `+1`, `-1`). On the facet line,
`U(t n) = D(t) = e^{-irt} P_+ + e^{irt} P_-` (rigidity). Levels `lambda`
are the projections `h.n`.

**Lemma A3 (first moments).** For every layer `lambda` and both signs,
`sum_{h in lambda} (h.e) <x_±, A_h x_±> = 0`.

*Proof (sign +).* `f(z; s) = e^{irt} <x_+, U(t n + s e) x_+>` is, for fixed
real `s`, an analytic polynomial in `z = e^{i delta t}` (the exponents
`r - h.n` are `>= 0` multiples of `delta`), with `|f| <= 1` on `|z| = 1`
(unitarity) and `f(z; 0) = 1` (rigidity). Let `g(z) = d f/ds (z; 0)`, again an
analytic polynomial. From `|f|^2 = 1 + 2 s Re g + O(s^2) <= 1` for both
signs of `s`, `Re g = 0` on the circle, so `g` is an imaginary constant.
At `z = 1` (`t = 0`), `g = -i <x_+, A(e) x_+> = -i r (e.n) = 0`. Hence
`g = 0`; its Fourier coefficients are `-i sum_{h in lambda} (h.e) <x_+, A_h x_+>`.
The sign `-` is the same with `e^{-irt}` and `z̄`. ∎

**Lemma A4 (off-diagonal first order).** For every layer `lambda`,
`sum_{h in lambda} (h.e) <x_+, A_h x_-> = sum_{h in -lambda} (h.e) conj<x_-, A_h x_+>`.

*Proof.* Differentiating `U^dag U = I` along `e` on the facet line gives
that `K(t) = D(t)^dag d_e U(t n)` is anti-Hermitian. With
`d_e U = -i sum_h (h.e) A_h e^{-it h.n}`, `K_{+-} = -i sum_h (h.e) <x_+, A_h x_->
e^{it(r - h.n)}` and `conj K_{-+} = i sum_h (h.e) conj<x_-, A_h x_+>
e^{it(r + h.n)}`. `K_{+-} + conj K_{-+} = 0` for all `t`; matching the
frequency `r - lambda = r + lambda'` gives `lambda' = -lambda` and the
stated identity. (The diagonal of `K` gives `Re g = 0` only; A3 is
stronger because it uses analyticity.) ∎

A3 and A4 are consequences of unitarity and rigidity; they are not new
assumptions. Their use is that they are linear.

CHECKS = []
SX, SY, I2 = m1.SX, m1.SY, m1.I2
R3 = sp.sqrt(3)


def record(label, ok, detail=""):
    CHECKS.append((label, bool(ok)))
    print(f"[{'PASS' if ok else 'FAIL'}] {label}" + (f" ({detail})" if detail else ""))


def eigvecs(n):
    """Unit +- eigenvectors of n.sigma for n in the xy plane at angle a."""
    a = sp.atan2(n[1], n[0])
    xp = sp.Matrix([1, sp.exp(sp.I * a)]) / sp.sqrt(2)
    xm = sp.Matrix([1, -sp.exp(sp.I * a)]) / sp.sqrt(2)
    return xp.applyfunc(sp.simplify), xm.applyfunc(sp.simplify)


def unknowns(npts):
    """Complex unknowns z (4 per point) and formal symbols w standing for
    conj(z); equations are linear in (z, w) and split into real form in
    (Re z, Im z) afterwards."""
    zs, ws, A, Abar = [], [], [], []
    for k in range(npts):
        z = sp.symbols(f"z{k}_0:4")
        w = sp.symbols(f"w{k}_0:4")
        zs.extend(z)
        ws.extend(w)
        A.append(sp.Matrix([[z[0], z[1]], [z[2], z[3]]]))
        Abar.append(sp.Matrix([[w[0], w[1]], [w[2], w[3]]]))
    return zs, ws, A, Abar


def linear_system(pts, normals, r, N, A, moments, Abar=None):
    """Linear equations (expressions == 0) in z and, for L3, w = conj(z)."""
    eqs = []
    for n in normals:
        L = m1.layers(pts, n, r, N)
        Pp = (I2 + n[0] * SX + n[1] * SY) / 2
        Pm = (I2 - n[0] * SX - n[1] * SY) / 2
        e = sp.Matrix([-n[1], n[0]])
        xp, xm = eigvecs(n)
        for lev, idx in L.items():
            target = Pp if lev == N else Pm if lev == -N else sp.zeros(2)
            M = sum((A[i] for i in idx), sp.zeros(2)) - target
            eqs += [sp.expand(x) for x in M]
            if moments:
                for x in (xp, xm):
                    s_ = sum((sp.nsimplify((pts[i].T * e)[0]) * (x.H * A[i] * x)[0] for i in idx), 0)
                    eqs.append(sp.expand(s_))
            if Abar is not None:
                # L3: off-diagonal anti-Hermiticity of D^dag d_e U on the line:
                # sum_{h in lev} (h.e) <x+, A_h x-> = sum_{h in -lev} (h.e) conj<x-, A_h x+>
                lhs = sum((sp.nsimplify((pts[i].T * e)[0]) * (xp.H * A[i] * xm)[0] for i in idx), 0)
                rhs = sum((sp.nsimplify((pts[i].T * e)[0]) * (xm.T * Abar[i] * xp.conjugate())[0]
                           for i in L.get(-lev, [])), 0)
                eqs.append(sp.expand(lhs - rhs))
    return [q for q in eqs if q != 0]


def real_form(eqs, zs, ws):
    """System C z + D conj(z) = b  ->  real system in (Re z, Im z)."""
    n = len(zs)
    CD, b = sp.linear_eq_to_matrix(eqs, list(zs) + list(ws))
    rows = []
    for i in range(CD.rows):
        re_row, im_row = [], []
        for j in range(n):
            c, d = sp.expand(CD[i, j]), sp.expand(CD[i, n + j])
            cr, ci = sp.expand(sp.re(c)), sp.expand(sp.im(c))
            dr, di = sp.expand(sp.re(d)), sp.expand(sp.im(d))
            re_row += [cr + dr, -ci + di]
            im_row += [ci + di, cr - dr]
        br, bi = sp.expand(sp.re(b[i])), sp.expand(sp.im(b[i]))
        rows += [re_row + [br], im_row + [bi]]
    return rows


def rank_and_free(eqs, zs, ws):
    rows = real_form(eqs, zs, ws)
    nv = 2 * len(zs)
    full = DomainMatrix.from_list_sympy(len(rows), nv + 1, rows, extension=True)
    coef = DomainMatrix.from_list_sympy(len(rows), nv, [r_[:nv] for r_ in rows], extension=True)
    rk, rka = coef.rank(), full.rank()
    return rk, nv - rk, rka


def parametrize(eqs, zs, ws, tname="t"):
    """Solve the real linear system exactly over QQ<sqrt3>; return the real
    and imaginary parts (P, Q) of each A_h as 2x2 matrices of affine forms in
    the free real parameters t."""
    rows = real_form(eqs, zs, ws)
    nv = 2 * len(zs)
    full = DomainMatrix.from_list_sympy(len(rows), nv + 1, rows, extension=True)
    R, pivots = full.rref()
    R = R.to_Matrix()
    free = [j for j in range(nv) if j not in pivots]
    t = sp.symbols(f"{tname}0:{len(free)}", real=True)
    val = {}
    for j, tj in zip(free, t):
        val[j] = tj
    for row, j in enumerate(pivots):
        val[j] = sp.expand(R[row, nv] - sum(R[row, f] * val[f] for f in free))
    P, Q = [], []
    for k in range(len(zs) // 4):
        P.append(sp.Matrix(2, 2, [val[2 * (4 * k + m)] for m in range(4)]))
        Q.append(sp.Matrix(2, 2, [val[2 * (4 * k + m) + 1] for m in range(4)]))
    return t, P, Q


def unitarity_equations(pts, P, Q):
    """Real polynomial equations of sum_{h-h'=d} A_h A_h'^dag = delta I and
    sum_{h-h'=d} A_h'^dag A_h = delta I, with A = P + iQ."""
    blocks = {}
    for i, hi in enumerate(pts):
        for j, hj in enumerate(pts):
            d = tuple(sp.nsimplify(c) for c in (hi - hj))
            b = blocks.setdefault(d, [sp.zeros(2) for _ in range(4)])
            # A_i A_j^dag = (P_i P_j^T + Q_i Q_j^T) + i (Q_i P_j^T - P_i Q_j^T)
            b[0] += P[i] * P[j].T + Q[i] * Q[j].T
            b[1] += Q[i] * P[j].T - P[i] * Q[j].T
            # A_j^dag A_i = (P_j^T P_i + Q_j^T Q_i) + i (P_j^T Q_i - Q_j^T P_i)
            b[2] += P[j].T * P[i] + Q[j].T * Q[i]
            b[3] += P[j].T * Q[i] - Q[j].T * P[i]
    eqs = set()
    for d, (re1, im1, re2, im2) in blocks.items():
        target = I2 if all(c == 0 for c in d) else sp.zeros(2)
        for M in (re1 - target, im1, re2 - target, im2):
            for e in M:
                e = sp.expand(e)
                if e != 0:
                    eqs.add(e)
    return sorted(eqs, key=sp.default_sort_key)


def gram_of(poly, t):
    """Unique symmetric Gram matrix of a quadratic polynomial in the basis (1, t)."""
    n = len(t)
    G = sp.zeros(n + 1)
    for mon, c in sp.Poly(poly, *t).terms():
        nz = [i for i, p_ in enumerate(mon) if p_]
        if not nz:
            G[0, 0] += c
        elif len(nz) == 1 and mon[nz[0]] == 1:
            G[0, nz[0] + 1] += c / 2
            G[nz[0] + 1, 0] += c / 2
        elif len(nz) == 1:
            G[nz[0] + 1, nz[0] + 1] += c
        else:
            i, j = nz
            G[i + 1, j + 1] += c / 2
            G[j + 1, i + 1] += c / 2
    return G


def positive_q3(x):
    """Exact sign test for x = a + b sqrt3, a, b rational."""
    x = sp.expand(x)
    b_ = x.coeff(R3)
    a_ = sp.expand(x - b_ * R3)
    if not (a_.is_Rational and b_.is_Rational):
        raise ValueError(f"not in Q(sqrt3): {x}")
    if a_ >= 0 and b_ >= 0:
        return a_ > 0 or b_ > 0
    if a_ > 0 and b_ < 0:
        return a_ ** 2 > 3 * b_ ** 2
    if a_ < 0 and b_ > 0:
        return 3 * b_ ** 2 > a_ ** 2
    return False


def ldl_pivots_positive(G):
    """Gaussian elimination without pivoting over QQ<sqrt3>; a real symmetric
    matrix is positive definite iff every pivot is positive (Sylvester)."""
    dm = DomainMatrix.from_list_sympy(G.rows, G.cols, G.tolist(), extension=True)
    K = dm.domain
    a = [list(row) for row in dm.to_list()]
    n = len(a)
    pivots = []
    for k in range(n):
        pk = a[k][k]
        if not positive_q3(K.to_sympy(pk)):
            return False, k, K.to_sympy(pk)
        pivots.append(pk)
        for i in range(k + 1, n):
            f = a[i][k] / pk
            if f != K.zero:
                for j in range(k, n):
                    a[i][j] = a[i][j] - f * a[k][j]
    return True, n, None


def check_certificate(t, eqs, path):
    """Exact real infeasibility certificate (contract Sec. 4 (ii) with
    constant multipliers): p = sum_i c_i f_i with rational c_i read from
    `path`; if the Gram matrix G of p in the basis (1, t) is positive
    definite then p(t) = (1,t) G (1,t)^T > 0 for every real t, while every
    f_i vanishes at a solution. In contract form: for 0 < mu <= 1/(G^-1)_00,
    G - mu e0 e0^T is PSD, = L D L^T with D >= 0, so p = mu + sum_j d_j l_j^2
    and -1 = sum_j (d_j/mu) l_j^2 + sum_i (-c_i/mu) f_i, a weighted sum of
    squares with exact coefficients. The verifier checks G > 0 exactly, which
    is what makes such a representation exist."""
    import json
    from fractions import Fraction
    data = json.loads(Path(path).read_text())
    c = [sp.Rational(Fraction(x).numerator, Fraction(x).denominator) for x in data["multipliers"]]
    record("certificate file matches the equation list", data["n_equations"] == len(eqs) == len(c))
    p = sp.expand(sum((ci * fi for ci, fi in zip(c, eqs) if ci != 0), sp.Integer(0)))
    G = gram_of(p, t)
    v = sp.Matrix([1] + list(t))
    record("p equals (1,t) G (1,t)^T exactly", sp.expand((v.T * G * v)[0] - p) == 0)
    ok, k, piv = ldl_pivots_positive(G)
    record("CERTIFICATE: the Gram matrix of p = sum c_i f_i is positive definite over QQ(sqrt3) "
           "(all 19 pivots > 0): p > 0 on R^18, so the reduced S19 system has no real solution",
           ok, "all pivots positive" if ok else f"pivot {k} not positive: {piv}")


def check_square_control():
    """Square substrate: U = exp(-ik_x s_x) exp(-ik_y s_y) has coefficients
    A_(hx,hy) = X_hx Y_hy at e^{-ik.h}, X_+-1 = (I +- s_x)/2,
    Y_+-1 = (I +- s_y)/2; it saturates c_0 = r_in = 1 with A(n) = n.sigma.
    Facet normals x, y. The walk must satisfy the layer sums and the
    first-moment equations (A3)."""
    X = {1: (I2 + SX) / 2, -1: (I2 - SX) / 2}
    Y = {1: (I2 + SY) / 2, -1: (I2 - SY) / 2}
    pts = [sp.Matrix([a, b]) for a in (1, -1) for b in (1, -1)]
    A = [X[a] * Y[b] for a in (1, -1) for b in (1, -1)]
    ok_sum = sp.simplify(sum(A, sp.zeros(2)) - I2) == sp.zeros(2)
    ok = True
    for n in (sp.Matrix([1, 0]), sp.Matrix([0, 1])):
        e = sp.Matrix([-n[1], n[0]])
        xp, xm = eigvecs(n)
        Pp = (I2 + n[0] * SX + n[1] * SY) / 2
        levels = {}
        for i, p in enumerate(pts):
            levels.setdefault((p.T * n)[0], []).append(i)
        ok = ok and sp.simplify(sum((A[i] for i in levels[1]), sp.zeros(2)) - Pp) == sp.zeros(2)
        for lev, idx in levels.items():
            for x in (xp, xm):
                s_ = sp.simplify(sum(((pts[i].T * e)[0] * (x.H * A[i] * x)[0] for i in idx), 0))
                ok = ok and s_ == 0
            lhs = sum(((pts[i].T * e)[0] * (xp.H * A[i] * xm)[0] for i in idx), 0)
            rhs = sum(((pts[i].T * e)[0] * sp.conjugate((xm.H * A[i] * xp)[0]) for i in levels.get(-lev, [])), 0)
            ok = ok and sp.simplify(lhs - rhs) == 0
    record("CONTROL square: the saturating walk satisfies the layer sums, A3 (L2) and the off-diagonal L3",
           ok_sum and ok)
    P = [Ak.applyfunc(sp.re) for Ak in A]
    Q = [Ak.applyfunc(sp.im) for Ak in A]
    record("CONTROL square: unitarity_equations of the unitary square walk are all identically 0",
           unitarity_equations(pts, P, Q) == [])
    P[0] = P[0] + sp.Rational(1, 7) * I2
    record("CONTROL square: a perturbed (non-unitary) coefficient is detected by unitarity_equations",
           unitarity_equations(pts, P, Q) != [])


def check_pipeline_square_control():
    """Positive control of the whole reduction pipeline (L1 + L2 + L3,
    parametrize, unitarity_equations) on the 3x3 square support, which, like
    S19, has interior and edge-interior points: the known saturating walk
    exp(-ik_x s_x) exp(-ik_y s_y) must lie in the parametrized family and
    satisfy every reduced equation."""
    pts = [sp.Matrix([a, b_]) for a in (-1, 0, 1) for b_ in (-1, 0, 1)]
    normals = [sp.Matrix([1, 0]), sp.Matrix([0, 1])]
    zs, ws, A, Abar = unknowns(len(pts))
    e3 = linear_system(pts, normals, sp.Integer(1), 1, A, moments=True, Abar=Abar)
    t, P, Q = parametrize(e3, zs, ws, tname="u")
    X = {1: (I2 + SX) / 2, -1: (I2 - SX) / 2}
    Y = {1: (I2 + SY) / 2, -1: (I2 - SY) / 2}
    walk = [X[int(p_[0])] * Y[int(p_[1])] if p_[0] != 0 and p_[1] != 0 else sp.zeros(2) for p_ in pts]
    match = []
    for k in range(len(pts)):
        match += list(P[k] - walk[k].applyfunc(sp.re)) + list(Q[k] - walk[k].applyfunc(sp.im))
    match = [sp.expand(m) for m in match if sp.expand(m) != 0]
    sol = sp.solve(match, t, dict=True) if match else [{}]
    ok = len(sol) >= 1 and all(sp.expand(m.subs(sol[0])) == 0 for m in match)
    record("CONTROL pipeline (3x3 square): the saturating walk lies in the L1+L2+L3 parametrized family",
           ok, f"{len(t)} free parameters")
    if ok:
        eqs = unitarity_equations(pts, P, Q)
        val = {ti: sol[0].get(ti, 0) for ti in t}
        record("CONTROL pipeline (3x3 square): it satisfies every reduced unitarity equation",
               all(sp.expand(e.subs(val)) == 0 for e in eqs), f"{len(eqs)} equations")

           free1 == 56)

    failed = [c for c in CHECKS if not c[1]]

    return 1 if failed else 0

## 1. L0 — the ceiling `c_0 <= r_in` on `W_hex`

Let `U` be in `W_hex`. Multiplying by the scalar `e^{i phi}` does not change
the support, so take `U(0) = I`. Let `n_f` be the unit outward normal of a
facet of the regular hexagon `C = conv F`, with edge vector `d in Lambda`

the hexagon) and are attained at vertices, which lie in `F`; hence
`h_C(n_f) = N delta` for an integer `N` and every `m_h` satisfies
`|m_h| <= N`.

For unit vectors `x, y in C^2` set `p(s) = <y, U(s n_f / delta) x>`. Then
`p(s) = sum_{|m|<=N} c_m e^{-i m s}` is a trigonometric polynomial of degree
`N` with `|p| <= 1` (U unitary). Bernstein's inequality (Lemma B below,
proved here) gives `|p'(0)| <= N`, i.e. `|<y, dU/dt(0) x>| <= N delta =
h_C(n_f)` along `n_f`. Taking the supremum over `x, y`,
`||A(n_f)|| <= h_C(n_f) = r_in`, where `A(n) = i dU(t n)/dt (0)`. `A(n_f)` is
Hermitian: differentiating `U U^dag = I` at `t = 0`, where `U = I`, shows
`dU/dt(0)` is anti-Hermitian. With `s = 2` and a traceless isotropic cone,
`A(n_f)^2 = c_0^2 I`, so the Hermitian `A(n_f)` has spectrum `{±c_0}` and

norm). Therefore **`c_0 <= r_in` for every walk in `W_hex`**. Only the
facet normal is needed, since `r_in = min h_C` is attained there.

**Lemma B (Bernstein, proved in-repo; repair R1 of the 2026-11-02 Grok
review).** If `p(s) = sum_{|m|<=N} c_m e^{-ims}` and `|p| <= 1` on `R`, then
`|p'(s_0)| <= N` for every `s_0`.

`R'(s_0) = Re(e^{i alpha} p'(s_0)) = |p'(s_0)|`. So it suffices to show
`R'(s_0) <= N` for real `R`.

*Real case.* Suppose `R'(s_0) > N`. Choose `0 < eps < 1` so small that

`C(s_0) = R_eps(s_0)` and `C'(s_0) = -N sin beta in (0, N]`, so
`D = C - R_eps` has `D(s_0) = 0`, `D'(s_0) < 0`; in particular `D` is not

`p'(0) = -i N`. Then `P(z) = z^N p` (with `z = e^{is}`) is an analytic
polynomial with `|P| = |p| <= 1` on the unit circle, hence `|P| <= 1` on
the closed disc by the maximum-modulus principle, `P(1) = 1` and
`P'(1) = 0`.
By Julia's lemma (classical; a non-constant `P` with `|P| <= 1` and a
boundary fixed point has angular derivative `P'(1) > 0`) `P` is constant,
so `p(s) = e^{-iNs}`; equivalently this is the equality case of
Bernstein's inequality, listed as prior art in PROJECT_INSTRUCTIONS §6.
By Cauchy–Schwarz `U(t n_f) x = e^{-i r_in t} x` for all `t`, and likewise
for the `-c_0` eigenvector. Hence along every facet-normal line
`U(t n_f) = e^{-i r_in t} P_+(n_f) + e^{i r_in t} P_-(n_f)`: only the two
extreme layers of `F` survive in that direction. In coefficients: the sum
of `A_h` over the top edge is `P_+(n_f)`, over the bottom edge
`P_-(n_f)`, and over every intermediate layer `0`.
Citation status: Julia's lemma and the Bernstein equality case are
classical; no primary source for them is in `sources/` (page
[UNVERIFIED]).

**Step 2: normal form.** By a Bloch rotation, a joint rotation of space
and Bloch sphere about `z`, and scale, take `vert(H) = {(cos k pi/3,
sin k pi/3)}` and `A(n) = r_in n . sigma` with `n` in the `xy` Bloch plane.
The three facet normals are at 30°, 90°, 150°.

**Step 3: certificate (verifier).** Along each facet normal the layers
are top edge (2 vertices), middle layer (2 vertices and `0`), bottom edge.
The nine layer equations are consistent, leave the coefficient `X_0` of
one vertex free, and force `A_0 = -I/2`. Imposing

**Positive control.** The same procedure on the square substrate

def cross(u, v):

def check_L0_grid():
    p, q, a, b = sp.symbols("p q a b", integer=True)
    d = p * G1 + q * G2
    h = a * G1 + b * G2

def check_T1_algebra():
    th, ell = sp.symbols("theta0 ell", positive=True)
    g12, g13, g23, g = sp.symbols("g12 g13 g23 g", real=True)
    a = [ell * sp.Matrix([sp.cos(th + 2 * sp.pi * j / 3), sp.sin(th + 2 * sp.pi * j / 3)]) for j in range(3)]
    G = sp.Matrix([[1, g12, g13], [g12, 1, g23], [g13, g23, 1]])
    M = sp.zeros(2, 2)
    for i in range(3):
        for j in range(3):
            M += G[i, j] * a[i] * a[j].T
    M = M.applyfunc(lambda e: sp.simplify(sp.expand_trig(e)))
    traceless = M - M.trace() / 2 * sp.eye(2)
    eqs = [sp.simplify(traceless[0, 0]), sp.simplify(traceless[0, 1])]
    # linear in (g12, g13, g23); solve for all theta0
    jac = sp.Matrix([[sp.diff(e, x) for x in (g12, g13, g23)] for e in eqs]).applyfunc(sp.simplify)
    const = [sp.simplify(e.subs({g12: 0, g13: 0, g23: 0})) for e in eqs]
    record("T1: traceless part is linear homogeneous in (g12, g13, g23)", all(c == 0 for c in const))
    minor = sp.simplify(sp.expand_trig(jac[:, :2].det()))
    record("T1: a 2x2 minor of the coefficient matrix equals the theta0-free constant "
           "sqrt3 l^4/2, so the rank is 2 for every theta0 (repair R2, Grok 2026-11-02)",
           sp.simplify(minor - R3 * ell ** 4 / 2) == 0, f"minor = {minor}")
    null = jac * sp.Matrix([1, 1, 1])
    record("T1: null space is spanned by (1,1,1): isotropy iff g12 = g13 = g23",
           all(sp.simplify(e) == 0 for e in null))
    Mg = M.subs({g12: g, g13: g, g23: g}).applyfunc(sp.simplify)
    record("T1: at equal overlaps, sum g_ij a_i a_j^T = (3/2) l^2 (1 - g) I",
           all(sp.simplify(e) == 0 for e in (Mg - sp.Rational(3, 2) * ell ** 2 * (1 - g) * sp.eye(2))))
    Gg = G.subs({g12: g, g13: g, g23: g})
    eig = Gg.eigenvals()
    record("T1: Gram eigenvalues are 1 + 2g (once) and 1 - g (twice): PSD iff -1/2 <= g <= 1",
           eig == {1 + 2 * g: 1, 1 - g: 2})
    c0_raw = sp.sqrt(sp.Rational(3, 2) * (1 - g)) * ell
    c0 = c0_raw / (2 * ell)          # longest step 2*ell
    r_in = R3 / 2
    ratio = sp.simplify(c0 / r_in)
    record("T1: c0/r_in = sqrt(1 - g)/sqrt2, maximal at g = -1/2 with value sqrt3/2",
           sp.simplify(ratio - sp.sqrt(1 - g) / sp.sqrt(2)) == 0 and sp.simplify(ratio.subs(g, -sp.Rational(1, 2)) - R3 / 2) == 0)
    record("T1: sqrt3/2 < 1 (the ceiling is not reached in T1)", bool(R3 / 2 < 1))
    record("T1: ADMP overlaps g = 1/3 give c0 = 1/2; g = -1/2 gives c0 = 3/4",
           sp.simplify(c0.subs(g, sp.Rational(1, 3)) - sp.Rational(1, 2)) == 0
           and sp.simplify(c0.subs(g, -sp.Rational(1, 2)) - sp.Rational(3, 4)) == 0)


def t1_walk(tvecs):
    """U(k) = prod_{j=3..1} exp(-i (k.a_j) tau_j), a_j = u_0, u_1, u_2 (unit,
    120 deg), tau_j = t_j . sigma, as a Laurent symbol in z = exp(-i k.g)."""
    zs = sp.symbols("z1:3")
    u = [(1, 0), (-1, 1), (0, -1)]          # u_0, u_1, u_2 in (g1, g2)
    factors = []
    for (tx, ty, tz), uj in zip(tvecs, u):
        tau = tx * SX + ty * SY + tz * SZ
        plus, minus = (I2 + tau) / 2, (I2 - tau) / 2
        factors.append(base.monomial(zs, uj) * plus + base.monomial(zs, base.neg(uj)) * minus)
    return sp.expand(factors[2] * factors[1] * factors[0]), zs


def check_witness():
    tv = [(sp.cos(2 * sp.pi * j / 3), sp.sin(2 * sp.pi * j / 3), 0) for j in range(3)]
    tv = [tuple(sp.nsimplify(x) for x in t) for t in tv]
    overlaps = [sum(a * b for a, b in zip(tv[i], tv[j])) for i, j in ((0, 1), (0, 2), (1, 2))]
    record("witness: Bloch vectors are unit, coplanar, pairwise overlap -1/2",
           all(sp.simplify(sum(x ** 2 for x in t) - 1) == 0 for t in tv)
           and all(sp.simplify(o + sp.Rational(1, 2)) == 0 for o in overlaps))
    symbol, zs = t1_walk(tv)
    adj = base.laurent_adjoint(symbol, zs)
    record("witness: symbol is unitary", base.is_zero_matrix(sp.expand(symbol * adj) - I2))
    support = base.laurent_support(symbol, zs)
    expected = {(-2, 0), (-2, 2), (0, -2), (0, 0), (0, 2), (2, -2), (2, 0)}
    record("witness: composite F = {0, +-2 u_j} (zero coefficient tolerated only at 0)",
           set(support) - {(0, 0)} == expected - {(0, 0)})
    gens = [(1, 0), (sp.Rational(1, 2), R3 / 2)]

           len(verts) == 6 and len(facets) == 6 and sp.simplify(r_in - R3 / 2) == 0)
    record("witness: every vertex coefficient is non-nilpotent",
           all(not base.nilpotent(support[keys[i]]) for i in verts))
    at0, A, tilt, c2, iso, herm = base.node_speeds(symbol, zs, gens, L, -1, 2)

def check_admp():

def check_T2_minimal():
    """Minimal-support subclass: F within {0} U vert(H), any s = 2 walk (no
    product form assumed). WLOG (Bloch rotation, joint spatial/Bloch
    rotation, scale): vertices v_k = (cos k pi/3, sin k pi/3), the isotropic
    linear term A(n) = r n.sigma with n in the xy Bloch plane. Saturation
    c_0 = r_in forces, along each facet normal n, U(t n) = e^{-irt} P_+ +
    e^{irt} P_- (report, Bernstein equality / Julia), i.e. layer sums:
    top edge -> P_+(n), bottom edge -> P_-(n), middle layer -> 0."""
    v = [sp.Matrix([sp.cos(sp.pi * k / 3), sp.sin(sp.pi * k / 3)]) for k in range(6)]
    normals = [sp.Matrix([sp.cos(a), sp.sin(a)]) for a in (sp.pi / 6, sp.pi / 2, 5 * sp.pi / 6)]
    layers = []
    for n in normals:
        proj = [sp.nsimplify((vk.T * n)[0]) for vk in v]
        top = [k for k in range(6) if proj[k] == R3 / 2]
        bot = [k for k in range(6) if proj[k] == -R3 / 2]
        mid = [k for k in range(6) if proj[k] == 0]
        layers.append((top, bot, mid))
    record("T2-min: along each facet normal the layers are {top edge, middle (2 vertices + 0), bottom edge}",
           all(len(t) == 2 and len(b) == 2 and len(m) == 2 for t, b, m in layers)
           and [l[0] for l in layers] == [[0, 1], [1, 2], [2, 3]])

    def proj(n, sign):
        return (I2 + sign * (n[0] * SX + n[1] * SY)) / 2
    Pp = [proj(n, 1).applyfunc(sp.nsimplify) for n in normals]
    Pm = [proj(n, -1).applyfunc(sp.nsimplify) for n in normals]
    # generic unknown coefficients for the 6 vertices and the origin
    syms = []
    def gen(name):
        e = sp.symbols(f"{name}_0:8", real=True)
        syms.extend(e)
        return sp.Matrix([[e[0] + sp.I * e[1], e[2] + sp.I * e[3]], [e[4] + sp.I * e[5], e[6] + sp.I * e[7]]])
    X = [gen(f"X{k}") for k in range(6)]
    Z = gen("Z")
    lin = []
    for (top, bot, mid), Pt, Pb in zip(layers, Pp, Pm):
        for M in (X[top[0]] + X[top[1]] - Pt, X[bot[0]] + X[bot[1]] - Pb, X[mid[0]] + X[mid[1]] + Z):
            for e in M:
                e = sp.expand(e)
                lin += [sp.expand(sp.re(e)), sp.expand(sp.im(e))]
    x0 = syms[:8]
    sol = sp.solve(lin, [s_ for s_ in syms if s_ not in x0], dict=True)
    record("T2-min: layer equations are consistent, with the X0 coefficient free", len(sol) == 1)
    sol = sol[0]
    Zs = Z.subs(sol)
    record("T2-min: layer equations force A_0 = -I/2", base.is_zero_matrix(Zs + I2 / 2))
    Xs = [Xk.subs(sol) for Xk in X]
    pts = [(v[k], Xs[k]) for k in range(6)] + [(sp.zeros(2, 1), Zs)]
    eqs = set()
    blocks = {}
    for h1, A1 in pts:
        for h2, A2 in pts:
            d = tuple(sp.nsimplify(c) for c in (h1 - h2))
            blocks.setdefault(d, [sp.zeros(2), sp.zeros(2)])
            blocks[d][0] += A1 * A2.H
            blocks[d][1] += A2.H * A1
    for d, (m1, m2) in blocks.items():
        target = I2 if all(c == 0 for c in d) else sp.zeros(2)
        for M in (m1 - target, m2 - target):
            for e in M:
                e = sp.expand(e)
                for part in (sp.expand(sp.re(e)), sp.expand(sp.im(e))):
                    if part != 0:
                        eqs.add(part)
    G = sp.groebner(list(eqs), *x0, order="lex")
    uni = [g for g in G.exprs if g.free_symbols == {x0[6]} and sp.degree(g, x0[6]) == 2]
    ok = False
    if uni:
        a2, a1, a0 = sp.Poly(uni[0], x0[6]).all_coeffs()
        disc = sp.simplify(a1 ** 2 - 4 * a2 * a0)
        ok = bool(disc < 0)
        print(f"  T2-min: Groebner element {uni[0]} = 0, discriminant {disc}")
    record("T2-min: unitarity + saturation imply a real quadratic with negative discriminant "
           "(no real solution): the ceiling is not attained with F in {0} U vert(H)", ok)


def check_T2_square_control():
    """Positive control for the saturation method: the square substrate
    F within {0} U {(+-1,+-1)}, facet normals x and y, A(n) = r n.sigma.
    The same layer + unitarity system must be consistent, and the known
    saturating walk exp(-i k_x sigma_x) exp(-i k_y sigma_y) must solve it."""
    verts = [(1, 1), (1, -1), (-1, 1), (-1, -1)]
    syms = []
    def gen(name):
        e = sp.symbols(f"{name}_0:8", real=True)
        syms.extend(e)
        return sp.Matrix([[e[0] + sp.I * e[1], e[2] + sp.I * e[3]], [e[4] + sp.I * e[5], e[6] + sp.I * e[7]]])
    X = {v: gen(f"Y{i}") for i, v in enumerate(verts)}
    Z = gen("W")
    Px, Pxm = (I2 + SX) / 2, (I2 - SX) / 2
    Py, Pym = (I2 + SY) / 2, (I2 - SY) / 2
    lin = []
    for M in (X[(1, 1)] + X[(1, -1)] - Px, X[(-1, 1)] + X[(-1, -1)] - Pxm,
              X[(1, 1)] + X[(-1, 1)] - Py, X[(1, -1)] + X[(-1, -1)] - Pym, Z):
        for e in M:
            e = sp.expand(e)
            lin += [sp.expand(sp.re(e)), sp.expand(sp.im(e))]
    free = syms[:8]
    sol = sp.solve(lin, [s_ for s_ in syms if s_ not in free], dict=True)[0]
    pts = [(sp.Matrix(v), X[v].subs(sol)) for v in verts] + [(sp.zeros(2, 1), Z.subs(sol))]
    eqs = set()
    blocks = {}
    for h1, A1 in pts:
        for h2, A2 in pts:
            d = tuple(h1 - h2)
            blocks.setdefault(d, [sp.zeros(2), sp.zeros(2)])
            blocks[d][0] += A1 * A2.H
            blocks[d][1] += A2.H * A1

    # with z^h <-> exp(-i k.h): factor = z_x P_x^+ + z_x^{-1} P_x^-, same for y
