#!/usr/bin/env python3
"""Exact certificate for the DISPERSION_L3_ROW gate.
Contract: theory/DISPERSION_L3_ROW_GATE.md (preregistration commit d3868fa).

Adds row L3 of C_ACROSS_LATTICES (standard d = 3 Weyl battery: cases A+, B+,
A-, B- of verify_weyl_dirac_gate.verify_case) to the dispersion corrections
table, with the parent's helpers imported read-only from
scripts/verify_dispersion_corrections_gate.py (no reviewed byte changed).
For comparison (hypotheses HL3b, HL3c) the parent's rows printed "L4"
(CJWW (5.12), walk1()) and L5a are recomputed here with the parent's own row
setup (verify_dispersion_corrections_gate.py l. 565-576).
Fails closed: any failed check or exception exits nonzero.
Run from the repository root:  python3 -B scripts/verify_dispersion_L3_row_gate.py
"""
import sys
import time
from pathlib import Path

import sympy as sp

ROOT = Path(__file__).resolve().parents[1]
sys.path.insert(0, str(ROOT / "scripts"))
import verify_dispersion_corrections_gate as D   # reviewed, read-only
import verify_c_across_lattices_gate as P        # reviewed, read-only
import verify_weyl_dirac_gate as WD              # reviewed, read-only
import verify_directional_saturation as DS       # reviewed, read-only

R3 = sp.sqrt(3)
START = time.time()
N = 4                                            # parent's degree for d = 3 rows
GENS3 = [(1, 0, 0), (0, 1, 0), (0, 0, 1)]
KS = sp.symbols("k1:4", real=True)
NS = sp.symbols("n1:4", real=True)


def analyse(label, symbol, zs, gens, sigma, results, groups):
    """The parent's generic s = 2 branch (Part 2), applied to one symbol."""
    support = P.laurent_support(symbol, zs)
    cart = {h: sum((a * sp.Matrix(g) for a, g in zip(h, gens)), sp.zeros(3, 1)) for h in support}
    L = sp.radsimp(sp.simplify(max((sp.sqrt((v.T * v)[0]) for v in cart.values()), key=lambda x: float(x))))
    print(f"-- {label} (s = {symbol.rows}, d = 3, scale = {L}, N = {N})  [elapsed {time.time() - START:.0f} s]",
          flush=True)
    Uk = D.taylor_symbol(support, gens, L, sigma, KS, N)
    s1 = D.trunc(Uk.trace(), KS, N)
    s2 = D.trunc(Uk.det(), KS, N)
    D.analyse_block(label, s1, s2, KS, NS, N, 1, results, 1 / R3)
    members, non_lattice = D.walk_group(symbol, zs, gens, D.point_group("cubic"))
    groups[label] = members
    print(f"  {label}: G_walk = {D.describe_group(members, 3)}; non-lattice skipped: {len(non_lattice)}")
    return support


def same_pair(x, y, results):
    return (D.czero(results[x]["beta"] - results[y]["beta"], NS)
            and D.czero(results[x]["alpha"] - results[y]["alpha"], NS))


def main():
    results, groups = {}, {}
    zs = sp.symbols("w1:4")
    cases = {}
    print("== Part 1: row L3 symbols imported read-only (contract Sec. 1)")
    for chi in (1, -1):
        sym, scalar = WD.published_symbol(3, chi, zs)
        s = "+" if chi == 1 else "-"
        cases[f"L3 A{s}"] = (sym, scalar)
        cases[f"L3 B{s}"] = (sym.T, scalar)
    C = [(z + 1 / z) / 2 for z in zs]
    S = [(z - 1 / z) / (2 * sp.I) for z in zs]
    for label, (sym, scalar) in cases.items():
        chi = 1 if "+" in label else -1
        D.record(f"{label}: unitary as a Laurent matrix",
                 P.is_zero_matrix(sym * P.laurent_adjoint(sym, zs) - sp.eye(2)))
        # HL3a
        D.record(f"{label}: det U = 1 (Laurent identity)", sp.expand(sp.together(sym.det() - 1)) == 0)
        D.record(f"{label}: tr U = 2 (ccc + chi sss), chi = {chi} (HL3a)",
                 sp.expand(sp.together(sym.trace() - 2 * (C[0] * C[1] * C[2] + chi * S[0] * S[1] * S[2]))) == 0)
        sup = P.laurent_support(sym, zs)
        D.record(f"{label}: Fourier support is {{+-1}}^3", set(sup) == {(a, b, c) for a in (1, -1)
                                                                       for b in (1, -1) for c in (1, -1)})

    print("== Part 2: node branch expansions (contract Sec. 2)")
    for label, (sym, _) in cases.items():
        analyse(label, sym, zs, GENS3, +1, results, groups)
    # Comparison rows with the parent's own setup (l. 565-576).
    analyse("parent L4 = CJWW (5.12)", D.laurent_matrix_from_p(DS.walk1()), DS.Z, GENS3, +1, results, groups)
    symbol, zsa, _, gens_a = P.row_L5a()
    analyse("parent L5a", symbol, zsa, gens_a, -1, results, groups)

    print("== Part 3: hypotheses (contract Sec. 3)")
    hb = all(same_pair(x, "parent L4 = CJWW (5.12)", results) and groups[x] == groups["parent L4 = CJWW (5.12)"]
             for x in ("L3 A+", "L3 B+"))
    print(f"  HL3b (A+, B+ share (beta, alpha) and G_walk with CJWW (5.12)): {'CONFIRMED' if hb else 'REFUTED'}")
    hc_sign = all(D.czero(results[x]["beta"] + results["L3 A+"]["beta"], NS)
                  and D.czero(results[x]["alpha"] - results["L3 A+"]["alpha"], NS) for x in ("L3 A-", "L3 B-"))
    hc_l5a = all(same_pair(x, "parent L5a", results) for x in ("L3 A-", "L3 B-"))
    hc = hc_sign and hc_l5a
    print(f"  HL3c (A-, B-: beta opposite to A+, same alpha: {hc_sign}; same (beta, alpha) as L5a: {hc_l5a}): "
          f"{'CONFIRMED' if hc else 'REFUTED'}")
    hd = all(results[x]["lowest_aniso"] for x in cases)
    print(f"  HL3d (lowest correction anisotropic in all four cases): {'CONFIRMED' if hd else 'REFUTED'}")
    he = all(results[x]["alpha0"].is_negative for x in cases)
    print(f"  HL3e (alpha_0 < 0 in all four cases): {'CONFIRMED' if he else 'REFUTED'}; "
          f"alpha_0 = {[results[x]['alpha0'] for x in cases]}")
    for x in cases:
        r = results[x]
        print(f"  {x}: lowest = {r['lowest']}; beta = {r['beta0']} + ({r['beta3']}) n1 n2 n3; "
              f"{r['pattern'].split('; ')[1]}")
    failed = [c for c in D.CHECKS if not c[1]]
    print(f"{len(D.CHECKS) - len(failed)}/{len(D.CHECKS)} checks passed (dispersion L3 row gate; "
          f"includes the parent's per-row checks of the comparison rows).")
    return 1 if failed else 0


if __name__ == "__main__":
    sys.exit(main())

    record_quiet = zero(lam0 ** 2 - s20)
    if not record_quiet:
        raise ValueError("node factor is not a double root at k = 0")
    x = trunc(s2 / s20 - 1, ks, N)
    phi = trunc(sp.I / 2 * compose(LOG1P, x, ks, N), ks, N)
    inv_s2 = trunc(compose(INV1P, x, ks, N) / s20, ks, N)
    g = trunc(s1 ** 2 * inv_s2 / 4, ks, N)
    v = trunc(2 * (1 - g), ks, N)             # 1 - cos(2 omega)
    omega2 = trunc(compose(ACOS2, v, ks, N) / 4, ks, N)
    return phi, omega2, lam0


def reduce_sphere(expr, ns):
    """Reduce a polynomial in n modulo |n|^2 = 1 (last variable eliminated)."""
    expr = sp.expand(expr)
    if expr == 0 or len(ns) == 1:
        return sp.expand(expr.subs(ns[0] ** 2, 1)) if len(ns) == 1 else expr
    last = ns[-1]
    rest = 1 - sum(n ** 2 for n in ns[:-1])
    poly = sp.Poly(expr, last)
    out = sp.Integer(0)
    for (j,), c in poly.terms():
        out += c * last ** (j % 2) * rest ** (j // 2)
    return sp.expand(out)


def ray_expansion(phi, omega2, ks, ns, N, divisor=1):
    """Upper node branch along k = t n, |n| = 1:
    eps_+ - eps_0 = (phi + omega)/divisor = sum_j E_j(n) t^j.
    Returns c0 and the dict {j: E_j(n)} (sphere-reduced), j = 1..N-1,
    together with the homogeneous parts Q_m(n) of omega^2 and Phi_m of phi."""
    sub = {k: T * n for k, n in zip(ks, ns)}
    om2_t = sp.expand(omega2.subs(sub))
    Q = {m: reduce_sphere(om2_t.coeff(T, m), ns) for m in range(0, N + 1)}
    ph_t = sp.expand(phi.subs(sub))
    Phi = {m: reduce_sphere(ph_t.coeff(T, m), ns) for m in range(0, N + 1)}
    c0sq = canon(Q[2])
    c0 = sp.radsimp(sp.sqrt(c0sq))
    x = sum((Q[m] / c0sq * T ** (m - 2) for m in range(3, N + 1)), sp.Integer(0))
    sq = compose(SQRT1P, x, [T], N - 2)
    om_t = sp.expand(c0 * T * sq)
    eps_t = sp.expand((om_t + ph_t) / divisor)
    E = {j: reduce_sphere(eps_t.coeff(T, j), ns) for j in range(1, N)}
    return c0 / divisor, E, Q, Phi


def group_velocity(phi, omega2, ks, ns, N, c0, divisor=1):
    """|grad eps_+| along k = t n to relative order t^(N-2): returns the
    coefficients gamma_j with |grad eps_+| = c0 [1 + sum_j gamma_j t^j]."""
    sub = {k: T * n for k, n in zip(ks, ns)}
    grad_om2 = [sp.expand(sp.diff(omega2, k).subs(sub)) for k in ks]
    grad_phi = [sp.expand(sp.diff(phi, k).subs(sub)) for k in ks]
    om2_t = sp.expand(omega2.subs(sub))
    c0full = c0 * divisor
    c0sq = c0full ** 2
    x = sp.expand(sum((reduce_sphere(om2_t.coeff(T, m), ns) / c0sq * T ** (m - 2)
                       for m in range(3, N + 1)), sp.Integer(0)))
    inv_sqrt = compose(SQRT1P, x, [T], N - 2)     # sqrt(1+x)
    inv_sqrt = compose(INV1P, sp.expand(inv_sqrt - 1), [T], N - 2)  # 1/sqrt(1+x)
    comps = []
    for a in range(len(ks)):
        g = sp.expand(grad_om2[a])
        # grad omega = grad omega2 / (2 omega), omega = c0full t sqrt(1+x)
        g_over_t = sp.expand(sum(reduce_sphere(g.coeff(T, m), ns) * T ** (m - 1) for m in range(1, N)))
        val = sp.expand(g_over_t * inv_sqrt / (2 * c0full) + grad_phi[a])
        comps.append(trunc(val, [T], N - 2))
    v2 = trunc(sum(c ** 2 for c in comps), [T], N - 2)
    v2 = sp.expand(sum(reduce_sphere(v2.coeff(T, m), ns) * T ** m for m in range(0, N - 1)))
    lead = canon(v2.coeff(T, 0))
    if not zero(lead - c0sq):
        raise ValueError(f"group speed at the node is not c0: {lead} vs {c0sq}")
    y = trunc(v2 / c0sq - 1, [T], N - 2)
    vg = compose(SQRT1P, y, [T], N - 2)
    return {j: reduce_sphere(sp.expand(vg).coeff(T, j), ns) for j in range(1, N - 1)}


# ------------------------------------------------------------ angular content
def fourier2(f, ns):
    """Trigonometric content of f(cos th, sin th): returns {m: (a_m, b_m)} with
    f = a_0 + sum_{m>0} a_m cos(m th) + b_m sin(m th), exactly."""
    sub = {ns[0]: (W + 1 / W) / 2, ns[1]: (W - 1 / W) / (2 * sp.I)}
    g = sp.expand(f.subs(sub))
    out = {}
    if g == 0:
        return {0: (sp.Integer(0), sp.Integer(0))}
    shift = W ** 40
    poly = sp.Poly(sp.expand(g * shift), W)
    c = {}
    for (e,), coeff in poly.terms():
        c[e - 40] = canon(coeff)
    ms = sorted({abs(m) for m in c})
    for m in ms:
        cp, cm = c.get(m, 0), c.get(-m, 0)
        if m == 0:
            out[0] = (canon(cp), sp.Integer(0))
        else:
            out[m] = (canon(cp + cm), canon(sp.I * (cp - cm)))
    return out


def dfact(n):
    return sp.Integer(1) if n <= 0 else sp.Integer(n) * dfact(n - 2)


def sphere_average(f, ns):
    """Exact average over the unit sphere S^2 of a polynomial in n."""
    f = sp.expand(f)
    if f == 0:
        return sp.Integer(0)
    poly = sp.Poly(f, *ns)
    out = sp.Integer(0)
    for exps, c in poly.terms():
        if any(e % 2 for e in exps):
            continue
        out += c * sp.prod([dfact(e - 1) for e in exps]) / dfact(sum(exps) + 1)
    return canon(out)


def cubic_content(f, ns):
    """f on S^2 = a0 + b4 K4 + b6 K6 + residual, with K4 = sum n^4 - 3/5 and K6
    the part of n1^2 n2^2 n3^2 orthogonal (L^2(S^2)) to 1 and K4. 1, K4, K6
    span the cubic (O_h) invariants of degree <= 6 on S^2. A zero residual
    certifies that f has only cubic-invariant content up to degree 6."""
    def avg(x):
        return sphere_average(reduce_sphere(x, ns), ns)
    K4 = reduce_sphere(sum(n ** 4 for n in ns) - sp.Rational(3, 5), ns)
    m6 = reduce_sphere(sp.prod([n ** 2 for n in ns]), ns)
    K6 = reduce_sphere(m6 - avg(m6) - avg(m6 * K4) / avg(K4 * K4) * K4, ns)
    a0 = avg(f)
    b4 = canon(avg(f * K4) / avg(K4 * K4))
    b6 = canon(avg(f * K6) / avg(K6 * K6))
    residual = reduce_sphere(f - a0 - b4 * K4 - b6 * K6, ns)
    return a0, b4, b6, residual


# ------------------------------------------------------------ symmetry group
def rot2(angle):
    return sp.Matrix([[sp.cos(angle), -sp.sin(angle)], [sp.sin(angle), sp.cos(angle)]])


def point_group(name):
    if name == "line":
        return {"+1": sp.Matrix([[1]]), "-1": sp.Matrix([[-1]])}
    F = sp.diag(1, -1)
    if name == "square":
        g = {}
        for j in range(4):
            g[f"R{90 * j}"] = rot2(sp.pi * j / 2)
            g[f"R{90 * j}F"] = rot2(sp.pi * j / 2) * F
        return g
    if name == "hexagon":
        g = {}
        for j in range(6):
            g[f"R{60 * j}"] = rot2(sp.pi * j / 3)
            g[f"R{60 * j}F"] = rot2(sp.pi * j / 3) * F
        return g
    if name == "cubic":
        g = {}
        for perm in itertools.permutations(range(3)):
            for signs in itertools.product((1, -1), repeat=3):
                m = sp.zeros(3)
                for i in range(3):
                    m[i, perm[i]] = signs[i]
                g[f"P{perm}S{signs}"] = m
        return g
    raise ValueError(name)


def lp_equal(a, b, zs):
    shift = sp.prod([z ** 12 for z in zs])
    diff = sp.expand((a - b) * shift)
    if diff == 0:
        return True
    return all(zero(c) for c in sp.Poly(diff, *zs).coeffs())


def power_traces(symbol, n):
    """tr(U^m), m = 1..n, exact Laurent polynomials."""
    out, M = [], sp.eye(symbol.rows)
    for _ in range(n):
        M = (M * symbol).applyfunc(sp.expand)
        out.append(sp.expand(M.trace()))
    return out


def walk_group(symbol, zs, gens, group):
    """Elements g of the substrate point group with charpoly U(g k) = charpoly U(k).
    In characteristic 0 the characteristic polynomial of an s x s matrix is
    determined by tr(U^m), m = 1..s (Newton identities), and tr(U(g k)^m) is
    tr(U^m) with the monomial substitution z^h -> z^(M h), M the integer
    matrix of g in generator coordinates (k.h is invariant)."""
    d = len(gens[0])
    G = sp.Matrix([[sp.Matrix(g)[i] for g in gens] for i in range(d)])
    Ginv = G.inv()
    traces = power_traces(symbol, symbol.rows)
    members, non_lattice = [], []
    for name, g in group.items():
        M = (Ginv * g.T * G).applyfunc(canon)
        if not all(x.is_integer for x in M):
            non_lattice.append(name)
            continue
        sub = {zs[i]: sp.prod([zs[j] ** int(M[j, i]) for j in range(d)]) for i in range(d)}
        if all(lp_equal(sp.expand(tr.subs(sub, simultaneous=True)), tr, zs) for tr in traces):
            members.append(name)
    return members, non_lattice


def truncated_charpoly(Uk, ks, N):
    """Coefficients (low to high in lambda) of det(lambda - U(k)) truncated at
    total degree N in k, from Newton's identities on truncated power traces."""
    n = Uk.rows
    p, M = [], sp.eye(n)
    for _ in range(n):
        M = (M * Uk).applyfunc(lambda e: trunc(e, ks, N))
        p.append(trunc(M.trace(), ks, N))
    e = [sp.Integer(1)]
    for m in range(1, n + 1):
        acc = sum(((-1) ** (i - 1) * e[m - i] * p[i - 1] for i in range(1, m + 1)), sp.Integer(0))
        e.append(trunc(acc / m, ks, N))
    coeffs = [sp.Integer(0)] * (n + 1)
    for m in range(n + 1):
        coeffs[n - m] = (-1) ** m * e[m]
    return coeffs


def describe_group(members, dim):
    rot = [m for m in members if not m.endswith("F")] if dim == 2 else None
    if dim == 1:
        return f"{{{', '.join(members)}}}"
    if dim == 2:
        refl = [m for m in members if m.endswith("F")]
        return f"{len(members)} elements: rotations {sorted(rot)}, reflections {len(refl)}"
    if len(members) == 48:
        return "48 of 48 signed permutations (O_h)"
    return f"{len(members)} of 48 signed permutations: {sorted(members)}"

    """Matrices of verify_directional_saturation (functions of p) as Laurent
    matrices in DS.Z with z_k = exp(i p_k)."""

def row_L3s4_symbol(a, zs):
    """A_a(k) of CURRENT_STATE Sec. 4.1 at theta = pi/4: W exp(-i q.Gamma) with
    q_i = k_i/sqrt3 and z_i <-> exp(-i q_i), i.e. generators e_i/sqrt3, sigma=-1.
    The radical b = sqrt(1-a^2) built inside core() is replaced by the symbol
    B_SYM, reduced with b^2 = 1 - a^2 (reduce_b)."""
    b, w, ts, gammas, rs = AA.core(a, sp.pi / 4)
    gammas = tuple(g.xreplace({b: B_SYM}) for g in gammas)
    if any(e.has(sp.sqrt(1 - a ** 2)) for g in gammas for e in g):
        raise ValueError("L3s4: sqrt(1-a^2) survived the substitution")
    U = w
    for z, gam in zip(zs, gammas):
        cosq = (z + 1 / z) / 2
        sinq = (1 / z - z) / (2 * sp.I)     # z = exp(-i q)
        U = U * (cosq * sp.eye(4) - sp.I * sinq * gam)
    return U.applyfunc(sp.expand)


def hensel_quadratic(coeffs, lam0, ks, N):
    """coeffs: monic polynomial sum_j coeffs[j] lambda^j (coeffs[-1] = 1) with
    entries polynomials in k of degree <= N. At k = 0, lam0 is a double root
    and the cofactor is coprime to (lambda - lam0)^2. Returns (s1, s2) with
    the lifted quadratic factor lambda^2 - s1 lambda + s2, exact to degree N."""
    n = len(coeffs) - 1
    at0 = {k: 0 for k in ks}
    P0 = sp.Poly(sum(canon(c.subs(at0)) * LAM ** j for j, c in enumerate(coeffs)), LAM)
    Q0 = sp.Poly(sp.expand((LAM - lam0) ** 2), LAM)
    R0, rem = sp.div(P0, Q0)
    if not all(zero(c) for c in rem.all_coeffs()):
        raise ValueError("lam0 is not a double root at k = 0")
    R0 = sp.Poly([canon(c) for c in R0.all_coeffs()], LAM)
    # Unknowns u = (q0, q1, r0, ..., r_{n-3}); linear map u -> Q0 * r + R0 * q.
    us = sp.symbols(f"u0:{n}")
    qpoly = sp.Poly(us[0] + us[1] * LAM, LAM)
    rpoly = sp.Poly(sum(us[2 + i] * LAM ** i for i in range(n - 2)), LAM)
    lin = (Q0 * rpoly + R0 * qpoly).as_expr()
    S = sp.Matrix([[sp.expand(lin).coeff(LAM, j).coeff(u) for u in us] for j in range(n)])
    Sinv = S.inv().applyfunc(canon)
    Qparts = {0: Q0.as_expr()}
    Rparts = {0: R0.as_expr()}
    for m in range(1, N + 1):
        Pm = sum(homog(c, ks, m) * LAM ** j for j, c in enumerate(coeffs))
        conv = sum((Qparts[i] * Rparts[m - i] for i in range(1, m)), sp.Integer(0))
        rhs_poly = sp.expand(Pm - conv)
        rhs = sp.Matrix([rhs_poly.coeff(LAM, j) for j in range(n)])
        sol = (Sinv * rhs).applyfunc(sp.expand)
        Qparts[m] = sol[0] + sol[1] * LAM
        Rparts[m] = sum(sol[2 + i] * LAM ** i for i in range(n - 2))
    Q = sp.expand(sum(Qparts.values()))
    q1 = sp.expand(Q.coeff(LAM, 1))
    q0 = sp.expand(Q.coeff(LAM, 0))
    return -q1, q0


def analyse_block(label, s1, s2, ks, ns, N, divisor, results, expected_c0):
    """From the node quadratic factor to the tabulated quantities."""
    phi, omega2, lam0 = node_from_quadratic(s1, s2, ks, N)
    c0, E, Q, Phi = ray_expansion(phi, omega2, ks, ns, N, divisor)
    dim = len(ks)
    record(f"{label}: omega^2 has no terms of degree 0 or 1 and its degree-2 part is c0^2 |k|^2",
           czero(Q[0], ns) and czero(Q[1], ns) and zero(Q[2] - c0 ** 2 * divisor ** 2))
    record(f"{label}: c0 = {c0} agrees with the parent table", zero(c0 - expected_c0))
    record(f"{label}: no linear phase (phi has no degree-1 part: untilted node)", czero(Phi[1], ns))
    odd = {m: czero(Q[m], ns) for m in range(3, N + 1, 2)}
    even_phi = {m: czero(Phi[m], ns) for m in range(2, N + 1)}
    beta = sp.expand(E[2] / c0)
    alpha = sp.expand(E[3] / c0)
    print(f"  {label}: node eigenvalue at k=0: {lam0}; eps_0 = {canon(sp.I * sp.log(lam0))}")
    print(f"  {label}: Q3 = {Q[3]}; Q4 = {Q[4]}")
    if N >= 6:
        print(f"  {label}: Q5 = {Q[5]}; Q6 = {Q[6]}")
    print(f"  {label}: phi: Phi2 = {Phi[2]}, Phi3 = {Phi[3]}, Phi4 = {Phi[4]}")
    print(f"  {label}: beta(n) = {beta}")
    print(f"  {label}: alpha(n) = {alpha}")
    gam = group_velocity(phi, omega2, ks, ns, N, c0, divisor)
    print(f"  {label}: |grad eps_+| / c0 - 1 = " + " + ".join(f"({g}) t^{j}" for j, g in gam.items()))
    info = dict(c0=c0, beta=beta, alpha=alpha, E=E, Q=Q, Phi=Phi, gamma=gam, dim=dim,
                odd_zero=all(odd.values()), phi_zero=all(even_phi.values()))
    if dim == 1:
        info["alpha0"] = canon(alpha)
        info["aniso"] = sp.Integer(0)
        info["pattern"] = f"constant alpha = {info['alpha0']}"
    elif dim == 2:
        fb = fourier2(beta, ns)
        fa = fourier2(alpha, ns)
        info["fourier_beta"], info["fourier_alpha"] = fb, fa
        info["alpha0"] = fa.get(0, (sp.Integer(0),))[0]
        info["aniso"] = {m: v for m, v in fa.items() if m > 0 and not (zero(v[0]) and zero(v[1]))}
        info["pattern"] = (f"beta: {fb}; alpha: {fa}")
        if N >= 6:
            f5 = fourier2(sp.expand(E[5] / c0), ns)
            info["fourier_E5"] = f5
            print(f"  {label}: E5/c0 (relative order t^4) harmonics: {f5}")
    else:
        a0, b4, b6, res = cubic_content(alpha, ns)
        info["alpha0"] = a0
        info["aniso"] = {4: b4, 6: b6} if not (zero(b4) and zero(b6)) else {}
        info["cubic_residual"] = res
        n123 = sp.prod(ns)
        b0 = sphere_average(beta, ns)
        b3 = canon(sphere_average(reduce_sphere(beta * n123, ns), ns)
                   / sphere_average(reduce_sphere(n123 ** 2, ns), ns))
        info["beta0"], info["beta3"] = b0, b3
        info["beta_residual"] = reduce_sphere(beta - b0 - b3 * n123, ns)
        info["pattern"] = (f"beta = {b0} + ({b3}) n1 n2 n3 [+ residual "
                           f"{'0' if czero(info['beta_residual'], ns) else 'nonzero'}]; "
                           f"alpha = {a0} + ({b4}) K4 + ({b6}) K6 [+ residual "
                           f"{'0' if czero(res, ns) else 'nonzero'}]")
    if dim == 1:
        info["lowest"] = "beta" if not czero(beta, ns) else ("alpha" if not czero(alpha, ns) else "none (to t^%d)" % (N - 2))
    else:
        info["lowest"] = "beta" if not czero(beta, ns) else ("alpha" if not czero(alpha, ns) else "none")
    if info["lowest"] == "beta":
        if dim == 2:
            info["lowest_aniso"] = any(m > 0 and not (zero(v[0]) and zero(v[1])) for m, v in info["fourier_beta"].items())
        else:
            info["lowest_aniso"] = not zero(info["beta3"]) or not czero(info["beta_residual"], ns)
    elif info["lowest"] == "alpha":
        info["lowest_aniso"] = bool(info["aniso"]) or (dim == 3 and not czero(info["cubic_residual"], ns))
    else:
        info["lowest_aniso"] = False
    print(f"  {label}: lowest nonvanishing correction: {info['lowest']}; anisotropic: {info['lowest_aniso']}")
    print(f"  {label}: lowest corrections: beta == 0: {czero(beta, ns)}; alpha0 = {info['alpha0']}; "
          f"angular content: {info['pattern']}")
    results[label] = info
    return info


def validity_scale_1d(s1, ks):
    """d = 1: smallest |k| > 0 where the two branches meet (cos omega = +-1)."""
    k = ks[0]
    sols = []
    for sgn in (1, -1):
        try:
            sol = sp.solveset(sp.Eq(s1 / 2, sgn), k, domain=sp.Interval.open(0, 4 * sp.pi))
            if isinstance(sol, sp.FiniteSet):
                sols += [s for s in sol if s.is_real]
        except Exception:
            pass
    return min(sols, key=lambda x: float(x)) if sols else None

def residual(expression):

def check_zero(label, expression):

def check_equal(label, actual, expected):

def laurent_adjoint(matrix, variables):
    return matrix.H.subs({variable: 1 / variable for variable in variables}, simultaneous=True)


def published_symbol(dimension, chirality, variables):
    cosines = [(variable + 1 / variable) / 2 for variable in variables]
    sines = [(variable - 1 / variable) / (2 * sp.I) for variable in variables]
    if dimension == 1:
        scalar = cosines[0]
        vector = (0, 0, sines[0])
    elif dimension == 2:
        cos_x, cos_y = cosines
        sin_x, sin_y = sines
        scalar = cos_x * cos_y
        vector = (sin_x * cos_y, cos_x * sin_y, sin_x * sin_y)
    else:
        cos_x, cos_y, cos_z = cosines
        sin_x, sin_y, sin_z = sines
        scalar = cos_x * cos_y * cos_z + chirality * sin_x * sin_y * sin_z
        vector = (
            sin_x * cos_y * cos_z - chirality * cos_x * sin_y * sin_z,
            -chirality * cos_x * sin_y * cos_z - sin_x * cos_y * sin_z,
            cos_x * cos_y * sin_z - chirality * sin_x * sin_y * cos_z,
        )
    matrix = scalar * IDENTITY
    for component, pauli in zip(vector, PAULI):
        matrix -= sp.I * component * pauli
    return matrix.applyfunc(sp.expand), sp.expand(scalar)


def rotation_product(dimension, chirality, variables):
    axes = (2,) if dimension == 1 else tuple(range(dimension))
    signs = (1, -chirality, 1) if dimension == 3 else (1,) * dimension
    matrix = IDENTITY
    for variable, axis, sign in zip(variables, axes, signs):
        positive = (IDENTITY - sign * PAULI[axis]) / 2
        negative = (IDENTITY + sign * PAULI[axis]) / 2
        matrix = matrix * (variable * positive + negative / variable)
    return matrix.applyfunc(sp.expand)


def fourier_support(matrix, variables):
    shift = sp.prod(variables)
    coefficients = {}
    for row in range(matrix.rows):
        for column in range(matrix.cols):
            polynomial = sp.Poly(sp.expand(shift * matrix[row, column]), *variables)
            for powers, coefficient in polynomial.terms():
                displacement = tuple(1 - power for power in powers)
                coefficients.setdefault(displacement, sp.zeros(matrix.rows, matrix.cols))
                coefficients[displacement][row, column] += coefficient
    return {
        displacement: coefficient
        for displacement, coefficient in coefficients.items()
        if any(sp.expand(entry) != 0 for entry in coefficient)
    }


def hull_geometry(support, dimension):
    vertices = [sp.Matrix(displacement) / sp.sqrt(dimension) for displacement in support]
    facets = set()
    for selected in itertools.combinations(vertices, dimension):
        differences = sp.Matrix(dimension - 1, dimension, lambda row, column:
                                selected[row + 1][column] - selected[0][column])
        normals = differences.nullspace()
        if len(normals) != 1:
            continue
        normal = normals[0]
        offset = normal.dot(selected[0])
        if offset == 0:
            continue
        normalized = normal / offset
        distances = [sp.simplify(normalized.dot(vertex) - 1) for vertex in vertices]
        if all(distance <= 0 for distance in distances):
            facets.add(tuple(normalized))
    if not facets:
        raise ValueError("Hull has no certified facets")
    inradius = min(sp.simplify(1 / sp.sqrt(sum(value**2 for value in facet)))
                   for facet in facets)
    maximum = max(sp.sqrt(vertex.dot(vertex)) for vertex in vertices)
    return inradius, maximum, len(facets)


def spatial_derivative(expression, variable, dimension):
    return sp.I * variable * sp.diff(expression, variable) / sp.sqrt(dimension)


def verify_case(dimension, chirality, transpose):
    suffix = ("+" if chirality == 1 else "-") if dimension == 3 else ""
    name = f"d={dimension} {'B' if transpose else 'A'}{suffix}"
    variables = sp.symbols(f"z0:{dimension}", real=True, nonzero=True)
    matrix, scalar = published_symbol(dimension, chirality, variables)
    product = rotation_product(dimension, chirality, variables)
    if transpose:
        matrix, product = matrix.T, product.T
    check_zero(f"{name}: published vs product", matrix - product)
    adjoint = laurent_adjoint(matrix, variables)
    check_zero(f"{name}: Weyl unitarity", matrix * adjoint - IDENTITY)
    check_zero(f"{name}: Weyl quadratic", matrix**2 - 2 * scalar * matrix + IDENTITY)
    check_zero(f"{name}: scalar trace", sp.trace(matrix) - 2 * scalar)
    support = fourier_support(matrix, variables)
    corners = set(itertools.product((-1, 1), repeat=dimension))
    check_equal(f"{name}: active support", set(support), corners)
    inradius, maximum, facets = hull_geometry(support, dimension)
    check_zero(f"{name}: inradius", inradius - 1 / sp.sqrt(dimension))
    check_zero(f"{name}: maximum support", maximum - 1)
    check_equal(f"{name}: facets", facets, 2 * dimension)
    origin = dict.fromkeys(variables, 1)
    check_zero(f"{name}: zero node", matrix.subs(origin) - IDENTITY)

               for variable in variables]

                   spatial_derivative(scalar, first_variable, dimension).subs(origin))
        for second, second_variable in enumerate(variables):
            expected = sp.Rational(int(first == second), dimension)
            check_zero(f"{name}: Clifford {first},{second}",
                       spatial[first] * spatial[second] + spatial[second] * spatial[first]
                       - 2 * expected * IDENTITY)
            hessian = spatial_derivative(
                spatial_derivative(scalar, first_variable, dimension), second_variable, dimension)
            check_zero(f"{name}: Hessian {first},{second}", hessian.subs(origin) + expected)
    speed = sp.sqrt(sp.simplify(sp.trace(spatial[0]**2) / 2))
    check_zero(f"{name}: massless matching", speed - inradius)
    dirac = sp.BlockMatrix([[HOP * matrix, sp.I * MASS * IDENTITY],

    dirac_adjoint = laurent_adjoint(dirac, variables)
    check_zero(f"{name}: Dirac unitarity", dirac * dirac_adjoint - sp.eye(4))
    check_zero(f"{name}: Dirac quadratic", dirac**2 - 2 * HOP * scalar * dirac + sp.eye(4))
    check_zero(f"{name}: Dirac trace", sp.trace(dirac) - 4 * HOP * scalar)

def verify_mass_coefficients():
    scalar_argument = sp.Symbol("u", real=True)
    gap = sp.Symbol("M", positive=True)
    phase = sp.acos(HOP * scalar_argument)
    prefactor = phase / sp.sqrt(1 - HOP**2 * scalar_argument**2)

    def at_origin(expression):
        expression = expression.subs(scalar_argument, 1).subs(HOP, sp.sqrt(1 - MASS**2))
        expression = sp.simplify(expression)
        return expression.subs(sp.acos(sp.sqrt(1 - MASS**2)), gap)

    hopping = sp.sqrt(1 - MASS**2)
    ratio = gap * hopping / MASS
    check_zero("fixed-mass logarithm prefactor", at_origin(prefactor) - gap / MASS)
    for dimension in (1, 2, 3):
        speed_squared = sp.Rational(1, dimension)
        phase_coefficient = at_origin(sp.diff(phase**2, scalar_argument)) * (-speed_squared / 2)
        check_zero(f"d={dimension}: phase curvature", phase_coefficient - speed_squared * ratio)
        mass_curvature = MASS * at_origin(sp.diff(prefactor, scalar_argument)) * (-speed_squared / 2)
        expected_curvature = speed_squared * hopping * (1 - ratio) / (2 * MASS)
        check_zero(f"d={dimension}: quadratic mass term", mass_curvature - expected_curvature)
        check_zero(f"d={dimension}: independent H-squared curvature",
                   speed_squared * ratio**2 + 2 * gap * mass_curvature - phase_coefficient)
    mass_ratio = ratio.subs(gap, sp.asin(MASS))
    check_zero("c_H small-mass series", sp.series(mass_ratio, MASS, 0, 6).removeO()
               - (1 - MASS**2 / 3 - 2 * MASS**4 / 15))
    check_zero("c_omega2 small-mass series", sp.series(sp.sqrt(mass_ratio), MASS, 0, 6).removeO()
               - (1 - MASS**2 / 6 - 29 * MASS**4 / 360))
    check_zero("small-mass ratio limit", sp.limit(mass_ratio, MASS, 0, dir="+") - 1)
    angle = sp.Symbol("angle", real=True)
    check_zero("strict-bound derivative identity",
               sp.diff(sp.sin(angle) - angle * sp.cos(angle), angle) - angle * sp.sin(angle))
    check_zero("strict-bound zero endpoint", (sp.sin(angle) - angle * sp.cos(angle)).subs(angle, 0))
